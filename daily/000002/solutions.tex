\begin{enumerate}
  \item
        \[
        \begin{gathered}
          a \equiv a' \pmod{n} \iff \exists i \in \mathbb{Z} \colon a' = a + in \\
          b \equiv b' \pmod{n} \iff \exists j \in \mathbb{Z} \colon b' = b + jn \\
          a' + b' \equiv a + in + b + jn \equiv a + b + n(i + j) \equiv a + b \pmod{n}
        \end{gathered}
        \]

        Therefore $a + b \equiv a' + b' \pmod{n}$.

        \[
        \begin{gathered}
          a'b' \equiv (a + in)(b + jn) \equiv ab + ajn + inb + ijn^{2} \equiv ab + n(aj + ib + ijn) \equiv ab \pmod{n}
        \end{gathered}
        \]

        Therefore $ab \equiv a'b' \pmod{n}$.
  \item
        \[
        \begin{gathered}
          \nexists 0^{-1}, 2^{-1}, 4^{-1}, 6^{-1} \pmod{8} \\
          1^{-1} \equiv 1 \pmod{8} \\
          3^{-1} \equiv 3 \pmod{8} \\
          5^{-1} \equiv 5 \pmod{8} \\
          7^{-1} \equiv 7 \pmod{8} \\
          \exists a^{-1} \pmod{8} \iff \gcd(a, 8) = 1
        \end{gathered}
        \]
  \item
        \[
        \begin{gathered}
        n={p_{1}^{a_{1}} \ldots p_{k}^{a_{k}}} \\
        \text{let } a = p_{1}, b = p_{1}^{a_{1} - 1}\prod_{i=2}^{k}{p_{i}^{a_{i}}} \\
        a \not\equiv 0, b \not\equiv 0 \pmod{n} \\
        ab \equiv n \equiv 0 \pmod{n} \\
        \end{gathered}
        \]

        Let's assume $\mathbb{Z}/n\mathbb{Z}$ is a field:
        \[
        \begin{gathered}
          \exists a^{-1} \colon a^{-1}a \equiv 1 \pmod{n} \\
          a^{-1}ab \equiv a^{-1}0 \pmod{n} \\
          (a^{-1}a) \equiv 1 \implies b \equiv 0 \pmod{n}
        \end{gathered}
        \]

        which is a contradiction since we've established previously that $b \not\equiv 0 \pmod{n}$, therefore $\mathbb{Z}/n\mathbb{Z}$ is not a field.
  \item
        \[
        \begin{gathered}
          [a] \neq [0], p \nmid a \implies \gcd(a, p) = 1 \implies ax + py = 1 \implies ax + py \equiv 1 \pmod{p} \implies \\
          ax \equiv 1 \pmod{p} \implies [x] \equiv {[a]}^{-1}
        \end{gathered}
        \]
  \item
        \[
        \begin{gathered}
          1^{4} \equiv 1 \pmod{5} \\
          2^{4} \equiv 1 \pmod{5} \\
          3^{4} \equiv 1 \pmod{5} \\
          4^{4} \equiv 1 \pmod{5} \\
        \end{gathered}
        \]

        Conj.: $p$ is prime, $p \nmid a \implies a^{p-1} \equiv 1 \pmod{p}$.
\end{enumerate}
