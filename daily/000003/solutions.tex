\begin{enumerate}
  \item
        \[
        \begin{gathered}
        f + g = t^3 + t^2 \\
        f * g = t^5 + t^4 + 1 \\
        f^2 = t^6 + t^2 + 1
        \end{gathered}
        \]
  \item
        From definition of unit:
        \[
        uv=1
        \]

        then

        \[
        \begin{gathered}
          \deg(uv)=0 \\
          \deg(uv) = \deg(u) + \deg(v), \deg(u) \geq 0, \deg(v) \geq 0 \implies \deg(u) = 0, \deg(v) = 0
        \end{gathered}
        \]

        Therefore units of $\mathbb{F}[t]$ are polynoms of form $f(t) = c$ where $c$ is a unit of $\mathbb{F}$, since $\mathbb{F}$ is a field and $c \neq 0$ and by definition of a field $c$ has an inverse. ${F[t]}^{\times}=F^{\times}=F \setminus {0}$
  \item
        \[
        \begin{gathered}
          t^2 = tt\\
          t^2 + 1 = (t - 1)(t + 1)\\
          t^2 + t = t(t + 1) \\
          f(t) = t^2 + t + 1, f(0)=1, f(1)=1  \text{ --- irreducible }
        \end{gathered}
        \]
  \item
        \[
        \begin{gathered}
        f \mid g \implies \exists h \colon g=fh \\
        \deg(g)=\deg(f) + \deg(h), \deg(f) \geq 0, \deg(h) \geq 0 \implies \deg(g) \geq \deg(f)
        \end{gathered}
        \]

        If $\deg(f) = \deg(g) = d$ then $\deg(h) = 0, h(t)=c$ and:
        \[
        \begin{gathered}
          f(t) = a_0 + a_1t + \cdots + a_{d}{t^d} \\
          g(t) = b_0 + b_1t + \cdots + b_{d}{t^d} \\
          \forall i: b_i = ca_i \\
          g \neq 0, f \neq 0 \implies c \neq 0 \text{ and } f, g \text{ --- associates }
        \end{gathered}
        \]
  \item
        Assume $f(t) = t$ is reducible, then $f=gh, \deg(g) < \deg(f), \deg(h) < \deg(f)$. Then since $\deg(f) = 1, \text{ then } \deg(g) = \deg(h) = 0, h \in \{0, 1\}, g \in \{0, 1\}$. $0 * 0 = 0 \neq t, 0 * 1 = 0 \neq t, 1 * 0 = 0 \neq t, 1 * 1 = 1 \neq t$. Exactly same logic applies to $f(t) = t + 1$.

        Therefore since $t \mid t^2 + t, (t + 1) \mid t^2 + t \text{ then } t \text{ and } (t + 1)$ are factors of $t^2 + t$.

        Since
        \[
        \deg f\in\{2,3\} \implies f\text{ reducible over }F \iff f\text{ has a root in }F.
        \]
        then because for $\mathbb{F}_2=\{0, 1\}$ and $f(t) = t^2 + t + 1$
        \[
        \begin{gathered}
          f(0) = 1 \\
          f(1) = 1
        \end{gathered}
        \]

        therefore $f(t)$ doesn't have a root, it's not reducible.
\end{enumerate}
